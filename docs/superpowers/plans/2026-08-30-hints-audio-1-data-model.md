# Hints & Audio — Sub-milestone 1: Data Model + Migration — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the `sections` / `steps` / `progress` / `audio` config shape (schema §2 of the spec), migrate existing `hintGroups` into `steps` on load, and re-point the Google Sheets `Hotkeys` mirror at `steps` instead of `hintGroups` — with zero visible behavior change to the current UI.

**Architecture:** Pure additions to `src/config-schema.js` (validation) and a new pure `src/config-migrate.js` (the `hintGroups → steps` conversion), wired into `src/config.js`'s `load()`. `src/sheets.js` gains a `steps`-based row builder and `syncHotkeysTab` switches to it. The operator console and config page keep reading `hintGroups` unchanged until later sub-milestones (5–7) replace them — this plan is backend-only.

**Tech Stack:** Node.js 22+, `node:sqlite` (already wired via `db`), `node:test`. No new npm dependency.

**Spec:** `docs/superpowers/specs/2026-08-30-hints-and-audio-design.md` — read §2 (data model), §2.3 (migration), §2.4 (schema), §11.1 (this sub-milestone's scope).

## Global Constraints

- Node built-ins only server-side; the sole npm runtime dependency remains `googleapis`, isolated in `src/sheets.js`. CommonJS (`require`/`module.exports`).
- `hintGroups` is left in `config.json` **untouched** after migration (one-release fallback) — do not delete it.
- Migration triggers only when `steps` is absent **and** `hintGroups` is present. Already-migrated or `hintGroups`-less configs are untouched (no-op, no write, no `config_history` row).
- IDs generated by migration: step `id` = `"step_" + (index+1)` (1-based, in `hintGroups` array order); hint `id` = `"<stepId>_h" + (index+1)` (1-based, in that group's `hints` array order). Deterministic — no randomness — so tests are reproducible.
- Section IDs must be unique among sections; step IDs unique among steps; hint IDs unique **across all hints in all steps** (one global hint-id namespace, since a Sheets mirror or future event references a bare hint id).
- Hint `type` accepts only `"text"` or `"audio"` — reject `"video"` explicitly (reserved, not built yet) with the message `steps[i].hints[j].type invalid (must be text or audio)`.
- Every new/changed function is exported for direct unit testing (no reaching into closures).

```js
function isStr(v) { return typeof v === 'string'; }
function isNum(v) { return typeof v === 'number' && Number.isFinite(v); }
function isInt(v) { return Number.isInteger(v); }

const ALLOWED_TOP_LEVEL = new Set([
  'roomName', 'game', 'sheets', 'hintGroups', 'banks',
  'plcs', 'signals', 'rules', 'profiles', 'sequences',
  'sections', 'steps', 'progress', 'audio',
]);

const HINT_TYPES = ['text', 'audio'];

function validateSectionsStepsProgressAudio(cfg, errors) {
  const sectionIds = new Set();
  if (cfg.sections !== undefined) {
    if (!Array.isArray(cfg.sections)) errors.push('sections must be an array');
    else cfg.sections.forEach((s, i) => {
      if (!isStr(s.id)) errors.push(`sections[${i}].id must be a string`);
      else {
        if (sectionIds.has(s.id)) errors.push(`duplicate section id: ${s.id}`);
        sectionIds.add(s.id);
      }
      if (!isStr(s.name)) errors.push(`sections[${i}].name must be a string`);
      if (!isNum(s.order)) errors.push(`sections[${i}].order must be a number`);
      if (s.note !== undefined && !isStr(s.note)) errors.push(`sections[${i}].note must be a string`);
    });
  }

  const hintIds = new Set();
  if (cfg.steps !== undefined) {
    if (!Array.isArray(cfg.steps)) errors.push('steps must be an array');
    else {
      const stepIds = new Set();
      cfg.steps.forEach((st, i) => {
        if (!isStr(st.id)) errors.push(`steps[${i}].id must be a string`);
        else {
          if (stepIds.has(st.id)) errors.push(`duplicate step id: ${st.id}`);
          stepIds.add(st.id);
        }
        if (!isStr(st.name)) errors.push(`steps[${i}].name must be a string`);
        if (!isNum(st.order)) errors.push(`steps[${i}].order must be a number`);
        if (st.sectionId !== undefined && st.sectionId !== null) {
          if (!isStr(st.sectionId)) errors.push(`steps[${i}].sectionId must be a string`);
          else if (!sectionIds.has(st.sectionId)) errors.push(`steps[${i}] references unknown section: ${st.sectionId}`);
        }
        if (st.hints !== undefined) {
          if (!Array.isArray(st.hints)) { errors.push(`steps[${i}].hints must be an array`); return; }
          st.hints.forEach((h, j) => {
            if (!isStr(h.id)) errors.push(`steps[${i}].hints[${j}].id must be a string`);
            else {
              if (hintIds.has(h.id)) errors.push(`duplicate hint id: ${h.id}`);
              hintIds.add(h.id);
            }
            if (!HINT_TYPES.includes(h.type)) errors.push(`steps[${i}].hints[${j}].type invalid (must be text or audio)`);
            if (h.type === 'text' && (!isStr(h.text) || h.text.trim() === ''))
              errors.push(`steps[${i}].hints[${j}].text must be a non-empty string`);
            if (h.type === 'audio' && (!isStr(h.mediaRef) || h.mediaRef.trim() === ''))
              errors.push(`steps[${i}].hints[${j}].mediaRef must be a non-empty string`);
            for (const k of ['label', 'color', 'icon', 'key'])
              if (h[k] !== undefined && !isStr(h[k])) errors.push(`steps[${i}].hints[${j}].${k} must be a string`);
            if (h.countsAsClue !== undefined && typeof h.countsAsClue !== 'boolean')
              errors.push(`steps[${i}].hints[${j}].countsAsClue must be a boolean`);
          });
        }
      });
    }
  }

  if (cfg.progress !== undefined) {
    if (typeof cfg.progress !== 'object' || cfg.progress === null) errors.push('progress must be an object');
    else if (cfg.progress.flags !== undefined) {
      if (!Array.isArray(cfg.progress.flags)) errors.push('progress.flags must be an array');
      else cfg.progress.flags.forEach((f, i) => { if (!isStr(f)) errors.push(`progress.flags[${i}] must be a string`); });
    }
  }

  if (cfg.audio !== undefined) {
    if (typeof cfg.audio !== 'object' || cfg.audio === null) errors.push('audio must be an object');
    else {
      if (cfg.audio.volume !== undefined && !isNum(cfg.audio.volume)) errors.push('audio.volume must be a number');
      if (cfg.audio.events !== undefined) {
        if (typeof cfg.audio.events !== 'object' || cfg.audio.events === null) errors.push('audio.events must be an object');
        else for (const [name, ev] of Object.entries(cfg.audio.events)) {
          if (typeof ev !== 'object' || ev === null) { errors.push(`audio.events.${name} must be an object`); continue; }
          if (ev.file !== undefined && !isStr(ev.file)) errors.push(`audio.events.${name}.file must be a string`);
          if (ev.enabled !== undefined && typeof ev.enabled !== 'boolean') errors.push(`audio.events.${name}.enabled must be a boolean`);
          if (name === 'midShow' && ev.enabled) {
            if (!isNum(ev.atSecondsRemaining) || ev.atSecondsRemaining <= 0)
              errors.push('audio.events.midShow.atSecondsRemaining must be a positive number when enabled');
          }
        }
      }
    }
  }
}
```

```js
// src/config-migrate.js
// Pure conversion of the old flat `hintGroups` config shape into the new
// sections/steps/progress/audio shape. No fs, no db — config.js wires the
// write-back. See docs/superpowers/specs/2026-08-30-hints-and-audio-design.md §2.3.

function defaultAudio(existingVolume) {
  return {
    volume: existingVolume != null ? existingVolume : 0.4,
    events: {
      start:     { file: '', enabled: false },
      loop:      { file: '', enabled: false },
      midShow:   { file: '', enabled: false, atSecondsRemaining: 120 },
      win:       { file: '', enabled: false },
      lose:      { file: '', enabled: false },
      clueChime: { file: '', enabled: false },
    },
  };
}

function migrateHintGroupsToSteps(cfg) {
  if (!cfg || typeof cfg !== 'object') return { cfg: {}, migrated: false };
  if (cfg.steps !== undefined || cfg.hintGroups === undefined) {
    return { cfg, migrated: false };
  }

  const groups = Array.isArray(cfg.hintGroups) ? cfg.hintGroups : [];
  const steps = groups.map((g, i) => {
    const stepId = 'step_' + (i + 1);
    const hints = (g.hints || []).map((h, j) => ({
      id: stepId + '_h' + (j + 1),
      type: 'text',
      text: h.text || '',
      key: h.key || '',
      countsAsClue: true,
    }));
    return {
      id: stepId,
      name: g.name || ('Group ' + (i + 1)),
      order: i + 1,
      sectionId: null,
      hints,
    };
  });

  const migrated = {
    ...cfg,
    sections: cfg.sections || [],
    steps,
    progress: cfg.progress || { flags: [] },
    audio: cfg.audio || defaultAudio(cfg.game && cfg.game.volume),
  };

  return { cfg: migrated, migrated: true };
}

module.exports = { migrateHintGroupsToSteps };
```

```js
const fs = require('node:fs');
const { validateConfig } = require('./config-schema');
const { migrateHintGroupsToSteps } = require('./config-migrate');

function createConfig({ path, db, now = () => Date.now() }) {
  let cache = {};

  function load() {
    try { cache = JSON.parse(fs.readFileSync(path, 'utf8')); }
    catch { cache = {}; }

    const { cfg: migratedCfg, migrated } = migrateHintGroupsToSteps(cache);
    cache = migratedCfg;

    if (migrated) {
      try {
        fs.writeFileSync(path, JSON.stringify(cache, null, 2));
        db.prepare('INSERT INTO config_history (ts, json) VALUES (?, ?)')
          .run(now(), JSON.stringify(cache));
      } catch { /* best-effort — the in-memory migrated cache is still returned */ }
    }

    return cache;
  }

  function save(obj) {
    const { ok, errors } = validateConfig(obj);
    if (!ok) return { ok, errors };
    try {
      fs.writeFileSync(path, JSON.stringify(obj, null, 2));
    } catch (e) {
      return { ok: false, errors: ['could not write config: ' + e.message] };
    }
    try { db.prepare('INSERT INTO config_history (ts, json) VALUES (?, ?)')
            .run(now(), JSON.stringify(obj)); } catch {}
    cache = obj;
    return { ok: true, errors: [] };
  }

  return { load, save, current: () => cache };
}

module.exports = { createConfig, validateConfig };
```

```js
// Rows for the "Hotkeys" reference tab, sourced from the new steps/hints model:
// A=Group(step name), B=Hint(label for audio / text for text, falling back to
// mediaRef when no label is set), C=Hotkey. Steps with no hints contribute nothing.
function buildHotkeysRowsFromSteps(steps) {
  const rows = [];
  (steps || []).forEach(step => {
    (step.hints || []).forEach(h => {
      if (!h) return;
      const hintText = h.type === 'audio' ? (h.label || h.mediaRef || '') : (h.text || '');
      if (!hintText) return;
      rows.push([step.name || '', hintText, h.key || '']);
    });
  });
  return rows;
}
```
